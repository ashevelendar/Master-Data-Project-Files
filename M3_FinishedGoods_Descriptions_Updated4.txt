Attribute VB_Name = "Module4"
Option Compare Database
Option Explicit

' Imports one SharePoint request into the local SharePointImportStage table using legacy ItemType codes.
' This module does NOT modify SourceDocumentData or any SharePoint records.
' Supports Item Release requests using the same detail-line structure as new items.
' Each line's own AX Item Code is copied into ExistingAXItemCode.
' Mapping: Finished Goods=FIN, Raw Material=ING, Packaging=PKG.
' Existing downstream post-processing remains responsible for applicable INT conversions.

Public Sub ImportSharePointRequestToStage_ClearStage()
    SharePointImportStepOK = False
    On Error GoTo ErrorHandler

    Dim db As DAO.Database
    Dim rsRequest As DAO.Recordset
    Dim rsSource As DAO.Recordset
    Dim rsStage As DAO.Recordset
    Dim requestInput As String
    Dim requestID As Long
    Dim requestType As String
    Dim whatType As String
    Dim companyValue As String
    Dim detailTable As String
    Dim sourceType As String
    Dim sql As String
    Dim itemTypeValue As String
    Dim importedCount As Long
    Dim savedErrNumber As Long
    Dim savedErrDescription As String

    requestInput = InputBox("Enter the SharePoint Request ID to import:", "SharePoint Import", "17578")
    If Len(Trim$(requestInput)) = 0 Then Exit Sub

    If Not IsPositiveWholeNumberLegacyTypes(requestInput) Then
        MsgBox "Please enter a positive whole-number Request ID.", vbExclamation, "Invalid Request ID"
        Exit Sub
    End If

    If CDbl(requestInput) > 2147483647# Then
        MsgBox "The Request ID is too large.", vbExclamation, "Invalid Request ID"
        Exit Sub
    End If

    requestID = CLng(requestInput)
    Set db = CurrentDb

    sql = "SELECT * FROM [Requests] WHERE [ID]=" & CStr(requestID)
    Set rsRequest = db.OpenRecordset(sql, dbOpenSnapshot)

    If rsRequest.EOF Then
        MsgBox "Request ID " & requestID & " was not found in the Requests list.", vbExclamation, "Request Not Found"
        GoTo CleanExit
    End If

    requestType = GetFieldTextLegacyTypes(rsRequest, "Request Type")
    whatType = GetFieldTextLegacyTypes(rsRequest, "What Type")
    companyValue = GetFieldTextLegacyTypes(rsRequest, "AX Bakery Number")

    ' Item Release is now allowed through staging.
    ' The existing AX code is read from each individual detail row below,
    ' not from the parent Requests record.

    detailTable = GetDetailTableLegacyTypes(whatType)

    If Len(detailTable) = 0 Then
        MsgBox "Request ID: " & requestID & vbCrLf & _
               "Request Type: " & requestType & vbCrLf & _
               "What Type: " & whatType & vbCrLf & vbCrLf & _
               "This request type is not supported. No staging records changed.", _
               vbExclamation, "Unsupported Request Type"
        GoTo CleanExit
    End If

    If UCase$(Trim$(whatType)) = "FINISHED GOODS" Then
        sourceType = "FG"
    Else
        sourceType = "RMPKG"
    End If

    rsRequest.Close
    Set rsRequest = Nothing

    sql = "SELECT * FROM [" & detailTable & "] WHERE [BatchID]=" & CStr(requestID)
    Set rsSource = db.OpenRecordset(sql, dbOpenSnapshot)

    If rsSource.EOF Then
        MsgBox "No item lines were found for Request ID " & requestID & _
               " in " & detailTable & "." & vbCrLf & _
               "Existing staging records have not been changed.", _
               vbExclamation, "No Matching Items"
        GoTo CleanExit
    End If

    ' Confirm source availability before clearing staged rows.
    rsSource.MoveLast
    importedCount = rsSource.RecordCount
    rsSource.MoveFirst

    ' This staging table is intended to hold one request at a time.
    ' Clear prior staged rows only after validating the request and source rows.
    db.Execute "DELETE FROM [SharePointImportStage]", dbFailOnError
    Set rsStage = db.OpenRecordset("SharePointImportStage", dbOpenDynaset)

    Do While Not rsSource.EOF
        rsStage.AddNew

        rsStage.Fields("RequestID").value = requestID
        rsStage.Fields("SourceType").value = sourceType
        SetStageText rsStage, "Company", companyValue

        If sourceType = "FG" Then
            ' Finished Goods uses the legacy validation code FIN.
            rsStage.Fields("ItemType").value = "FIN"

            ' Build the Finished Goods AX description using the FG / SW / AL naming rules.
            ' The staging ItemType remains FIN for all three Finished Goods variants.
            SetStageText rsStage, "ItemDesc", BuildFinishedGoodsDescriptionLegacyTypes( _
                rsSource, GetFieldTextLegacyTypes(rsSource, "Proposed Item Description"))
            SetStageText rsStage, "SBCustName", GetFieldTextLegacyTypes(rsSource, "SB Customer Name")
            SetStageText rsStage, "VATGroup", GetFieldTextLegacyTypes(rsSource, "Selling VAT Group")
            SetStageText rsStage, "Warehouse", GetFieldTextLegacyTypes(rsSource, "Warehouse")
            SetStageText rsStage, "StockedUOM", GetFieldTextLegacyTypes(rsSource, "AX Selling Unit")

            ' Preserve the existing AX code belonging to THIS detail line.
            SetStageText rsStage, "ExistingAXItemCode", CleanExistingAXCode( _
    GetFieldTextLegacyTypes(rsSource, "AX Item Code"))

            SetStageText rsStage, "Sys21ItemCode", GetFieldTextLegacyTypes(rsSource, "Sys 21 Item Code")
            SetStageText rsStage, "AXItemCat", GetFieldTextLegacyTypes(rsSource, "AX Item Category")
            SetStageText rsStage, "AXProcCat", GetFieldTextLegacyTypes(rsSource, "AX Sales Category")
            SetStageText rsStage, "AXItemGroup", GetFieldTextLegacyTypes(rsSource, "AX Item Group")
            SetStageText rsStage, "ItemPackSize", CleanNumericValue( _
    GetFieldTextLegacyTypes(rsSource, "Case Size"))

SetStageText rsStage, "NetWeight", CleanNumericValue( _
    GetFieldTextLegacyTypes(rsSource, "Net Weight per Case"))
            SetStageText rsStage, "SellingPrice", GetFieldTextLegacyTypes(rsSource, "AX Selling Price Per Unit")
            SetStageText rsStage, "CustGroup", GetFieldTextLegacyTypes(rsSource, "Customer Group")
            SetStageText rsStage, "EDIItem", GetFieldTextLegacyTypes(rsSource, "EDI Item")
            SetStageText rsStage, "GTIN", GetFieldTextLegacyTypes(rsSource, "GTIN")
            SetStageText rsStage, "FromDate", GetFieldDateTextLegacyTypes(rsSource, "From Date")
            SetStageText rsStage, "ToDate", GetFieldDateTextLegacyTypes(rsSource, "To Date")
            SetStageText rsStage, "ReceivingDepot", GetFieldTextLegacyTypes(rsSource, "Receiving Depot")

        Else
            ' Do not use the SharePoint Item Type field; it may appear as 0 in Access.
            ' Classify from the Requests list's What Type instead.
            Select Case UCase$(Trim$(whatType))
                Case "RAW MATERIAL", "RAW MATERIALS", "INGREDIENT", "INGREDIENTS"
                    itemTypeValue = "ING"
                Case "PACKAGING"
                    itemTypeValue = "PKG"
                Case Else
                    MsgBox "Unsupported What Type '" & whatType & _
                           "'. No further rows were staged.", _
                           vbExclamation, "Unsupported Request Type"
                    GoTo CleanExit
            End Select

            rsStage.Fields("ItemType").value = itemTypeValue

            If itemTypeValue = "ING" Then
                SetStageText rsStage, "ItemDesc", BuildRawMaterialDescription( _
                    rsSource, GetFieldTextLegacyTypes(rsSource, "Proposed Item Description"))
            Else
                ' Packaging descriptions remain unchanged.
                SetStageText rsStage, "ItemDesc", GetFieldTextLegacyTypes(rsSource, "Proposed Item Description")
            End If
            SetStageText rsStage, "SBCustName", GetFieldTextLegacyTypes(rsSource, "SB Customer")
            SetStageText rsStage, "VATGroup", GetFieldTextLegacyTypes(rsSource, "VAT Group")
            SetStageText rsStage, "SupplierItemDesc", GetFieldTextLegacyTypes(rsSource, "Supplier Item Description As Per Spec")
            SetStageText rsStage, "SuppItemCode", GetFieldTextLegacyTypes(rsSource, "Supplier Item Code As Per Spec")
            SetStageText rsStage, "Site", GetFieldTextLegacyTypes(rsSource, "Site")
            SetStageText rsStage, "Warehouse", GetFieldTextLegacyTypes(rsSource, "Warehouse")
            SetStageText rsStage, "AX_Location", GetFieldTextLegacyTypes(rsSource, "Receipt Location")
            SetStageText rsStage, "StockedUOM", GetFieldTextLegacyTypes(rsSource, "Stocked Unit of Measure")

            ' Preserve the existing AX code belonging to THIS detail line.
            SetStageText rsStage, "ExistingAXItemCode", GetFieldTextLegacyTypes(rsSource, "AX Item Code")

            SetStageText rsStage, "Sys21ItemCode", GetFieldTextLegacyTypes(rsSource, "Sys 21 Item Code")
            SetStageText rsStage, "Sys21SuppCode", GetFieldTextLegacyTypes(rsSource, "Sys 21 Supplier Code")
            SetStageText rsStage, "AXItemCat", GetFieldTextLegacyTypes(rsSource, "AX Item Category")
            SetStageText rsStage, "AXProcCat", GetFieldTextLegacyTypes(rsSource, "AX Procurement Category")
            SetStageText rsStage, "AXItemGroup", GetFieldTextLegacyTypes(rsSource, "AX Item Group")
            SetStageText rsStage, "AXSuppACNum", GetFieldTextLegacyTypes(rsSource, "AX Supplier A/C Number")
            SetStageText rsStage, "AX SuppName", GetFieldTextLegacyTypes(rsSource, "AX Supplier Name")
            SetStageText rsStage, "AXPurchUnit", GetFieldTextLegacyTypes(rsSource, "AX Purchase Unit")
            SetStageText rsStage, "AXPurchPricePerUnit", GetFieldTextLegacyTypes(rsSource, "AX Purchase Price per Unit")
        End If

        rsStage.Update
        rsSource.MoveNext
    Loop

    rsStage.Close
    Set rsStage = Nothing
    rsSource.Close
    Set rsSource = Nothing

    SharePointImportStepOK = True

CleanExit:
    On Error Resume Next
    If Not rsStage Is Nothing Then rsStage.Close
    If Not rsSource Is Nothing Then rsSource.Close
    If Not rsRequest Is Nothing Then rsRequest.Close

    Set rsStage = Nothing
    Set rsSource = Nothing
    Set rsRequest = Nothing
    Set db = Nothing
    Exit Sub

ErrorHandler:
    savedErrNumber = Err.Number
    savedErrDescription = Err.Description

    On Error Resume Next
    If Not rsStage Is Nothing Then rsStage.Close
    Set rsStage = Nothing
    If Not rsSource Is Nothing Then rsSource.Close
    If Not rsRequest Is Nothing Then rsRequest.Close

    MsgBox "The SharePoint import failed." & vbCrLf & vbCrLf & _
           "Error " & savedErrNumber & ": " & savedErrDescription & vbCrLf & vbCrLf & _
           "If this failed during insertion, inspect SharePointImportStage before retrying.", _
           vbExclamation, "SharePoint Import Error"

    Resume CleanExit
End Sub


Private Function GetDetailTableLegacyTypes(ByVal whatType As String) As String
    Select Case UCase$(Trim$(whatType))
        Case "FINISHED GOODS"
            GetDetailTableLegacyTypes = "Item Request - FG"

        Case "PACKAGING", "RAW MATERIALS", "RAW MATERIAL", "INGREDIENT", "INGREDIENTS"
            GetDetailTableLegacyTypes = "Item Request - RMPKG"

        Case Else
            GetDetailTableLegacyTypes = vbNullString
    End Select
End Function


Private Function IsPositiveWholeNumberLegacyTypes(ByVal valueText As String) As Boolean
    Dim numberValue As Double

    If Len(Trim$(valueText)) = 0 Then Exit Function
    If Not IsNumeric(valueText) Then Exit Function

    numberValue = CDbl(valueText)
    IsPositiveWholeNumberLegacyTypes = (numberValue >= 1 And numberValue = Fix(numberValue))
End Function


Private Function GetFieldTextLegacyTypes(ByVal rs As DAO.Recordset, ByVal fieldName As String) As String
    Dim fieldValue As Variant

    On Error GoTo FieldMissing
    fieldValue = rs.Fields(fieldName).value

    If IsNull(fieldValue) Or IsEmpty(fieldValue) Then
        GetFieldTextLegacyTypes = vbNullString
    Else
        GetFieldTextLegacyTypes = CStr(fieldValue)
    End If

    Exit Function

FieldMissing:
    Err.Clear
    GetFieldTextLegacyTypes = vbNullString
End Function


Private Function GetFieldDateTextLegacyTypes(ByVal rs As DAO.Recordset, ByVal fieldName As String) As String
    Dim fieldValue As Variant

    On Error GoTo FieldMissing
    fieldValue = rs.Fields(fieldName).value

    If IsNull(fieldValue) Or IsEmpty(fieldValue) Then
        GetFieldDateTextLegacyTypes = vbNullString
    ElseIf IsDate(fieldValue) Then
        GetFieldDateTextLegacyTypes = Format$(CDate(fieldValue), "dd/mm/yyyy")
    Else
        GetFieldDateTextLegacyTypes = CStr(fieldValue)
    End If

    Exit Function

FieldMissing:
    Err.Clear
    GetFieldDateTextLegacyTypes = vbNullString
End Function


Private Sub SetStageText(ByVal rs As DAO.Recordset, ByVal fieldName As String, ByVal textValue As String)
    ' For text fields, preserve supplied values.
    ' If blank, store Null rather than a zero-length string (Access error 3315).
    If Len(Trim$(textValue)) = 0 Then
        rs.Fields(fieldName).value = Null
    Else
        rs.Fields(fieldName).value = textValue
    End If
End Sub



Private Function CleanExistingAXCode(ByVal rawValue As String) As String
    Dim i As Long
    Dim ch As String

    rawValue = Trim$(rawValue)

    ' AX codes must be exactly 6 characters.
    If Len(rawValue) <> 6 Then Exit Function

    ' AX codes must start with 1, 2 or 3.
    If Left$(rawValue, 1) < "1" Or Left$(rawValue, 1) > "3" Then Exit Function

    ' Every character must be a digit.
    For i = 1 To Len(rawValue)
        ch = Mid$(rawValue, i, 1)

        If ch < "0" Or ch > "9" Then Exit Function
    Next i

    CleanExistingAXCode = rawValue
End Function


Private Function CleanNumericValue(ByVal rawValue As String) As String
    rawValue = Trim$(rawValue)

    If Len(rawValue) = 0 Then Exit Function

    If IsNumeric(rawValue) Then
        CleanNumericValue = rawValue
    End If
End Function

Private Function BuildFinishedGoodsDescriptionLegacyTypes(ByVal rsSource As DAO.Recordset, ByVal originalDescription As String) As String

    Dim itemType As String
    Dim customerCode As String
    Dim customerName As String
    Dim descriptionText As String
    Dim result As String
    Dim parsedCode As String
    Dim parsedName As String

    itemType = GetFinishedGoodsItemTypeLegacyTypes(rsSource)

    ' Customer display values may look like "BKR - Booker" or "12345 BUD - Budgens".
    ' Extract the final short code before the hyphen and the customer name after it.
    customerName = GetFirstNonBlankFieldTextLegacyTypes(rsSource, _
        Array("SB Customer Name", "SB Customer", "Customer Name", "Customer"))
    ParseFinishedGoodsCustomerLegacyTypes customerName, parsedCode, parsedName
    If Len(parsedName) > 0 Then customerName = parsedName

    descriptionText = Trim$(originalDescription)

    Select Case itemType
        Case "ASSEMBLY LEVEL"
            ' Keep the customer abbreviation already present in the proposed description.
            ' Do not insert the full SB customer name (for example, Booker).
            descriptionText = RemoveLeadingTokenLegacyTypes(descriptionText, "Assy")
            result = "Assy"

        Case "SANDWICH LEVEL"
            ' Keep the customer abbreviation already present in the proposed description.
            ' Do not insert the full SB customer name (for example, Booker).
            descriptionText = RemoveLeadingTokenLegacyTypes(descriptionText, "SW")
            result = "SW"

        Case Else
            ' Standard Finished Goods: customer code + description only.
            customerCode = CleanFinishedGoodsCustomerCodeLegacyTypes(parsedCode)
            If Len(customerCode) > 0 Then
                descriptionText = RemoveLeadingTokenLegacyTypes(descriptionText, customerCode)
            End If
            result = customerCode
    End Select

    result = AppendFinishedGoodsPartLegacyTypes(result, descriptionText)

    ' AX permits a maximum of 62 characters. Trim only from the end.
    If Len(result) > 62 Then result = Left$(result, 62)

    BuildFinishedGoodsDescriptionLegacyTypes = Trim$(result)

End Function


Private Function GetFinishedGoodsItemTypeLegacyTypes(ByVal rsSource As DAO.Recordset) As String

    Dim itemType As String

    ' Classification is based exclusively on the SharePoint Item Type field.
    itemType = UCase$(Trim$(GetFirstNonBlankFieldTextLegacyTypes(rsSource, _
        Array("Item Type", "ItemType"))))

    If InStr(1, itemType, "ASSEMBLY LEVEL", vbTextCompare) > 0 Or itemType = "ASSY" Then
        GetFinishedGoodsItemTypeLegacyTypes = "ASSEMBLY LEVEL"
    ElseIf InStr(1, itemType, "SANDWICH LEVEL", vbTextCompare) > 0 Or itemType = "SW" Then
        GetFinishedGoodsItemTypeLegacyTypes = "SANDWICH LEVEL"
    Else
        GetFinishedGoodsItemTypeLegacyTypes = "FINISHED GOODS"
    End If

End Function


Private Sub ParseFinishedGoodsCustomerLegacyTypes(ByVal rawCustomer As String, ByRef customerCode As String, ByRef customerName As String)

    Dim parts() As String
    Dim leftPart As String
    Dim tokens() As String

    customerCode = vbNullString
    customerName = Trim$(rawCustomer)
    If Len(customerName) = 0 Then Exit Sub

    If InStr(1, customerName, " - ", vbTextCompare) > 0 Then
        parts = Split(customerName, " - ")
        leftPart = Trim$(parts(0))
        customerName = Trim$(Mid$(rawCustomer, InStr(1, rawCustomer, " - ", vbTextCompare) + 3))
        tokens = Split(leftPart, " ")
        If UBound(tokens) >= 0 Then
            customerCode = Trim$(tokens(UBound(tokens)))
        End If
    End If

End Sub


Private Function RemoveLeadingTokenLegacyTypes(ByVal textValue As String, ByVal token As String) As String

    textValue = Trim$(textValue)
    token = Trim$(token)

    If Len(token) > 0 And Len(textValue) >= Len(token) Then
        If StrComp(Left$(textValue, Len(token)), token, vbTextCompare) = 0 Then
            If Len(textValue) = Len(token) Then
                textValue = vbNullString
            ElseIf Mid$(textValue, Len(token) + 1, 1) = " " Then
                textValue = Trim$(Mid$(textValue, Len(token) + 1))
            End If
        End If
    End If

    RemoveLeadingTokenLegacyTypes = textValue

End Function


Private Function GetFinishedGoodsCostGroupLegacyTypes(ByVal rsSource As DAO.Recordset) As String

    Dim costGroup As String
    Dim ItemType As String
    Dim AXItemGroup As String

    ' Prefer the explicit Cost Group field if the linked SharePoint table exposes it.
    costGroup = UCase$(Trim$(GetFirstNonBlankFieldTextLegacyTypes(rsSource, _
        Array("Cost Group", "CostGroup", "AX Cost Group", "AX CostGroup", "Item Cost Group"))))

    Select Case costGroup
        Case "FG", "FG - FINISHED GOODS", "FINISHED GOODS", "FINISHED GOOD"
            GetFinishedGoodsCostGroupLegacyTypes = "FG"
            Exit Function
        Case "AL", "AL - ASSEMBLY LEVEL", "ASSEMBLY LEVEL", "ASSEMBLY", "ASSY"
            GetFinishedGoodsCostGroupLegacyTypes = "AL"
            Exit Function
        Case "SW", "SW - SANDWICH LEVEL", "SANDWICH LEVEL", "SANDWICH"
            GetFinishedGoodsCostGroupLegacyTypes = "SW"
            Exit Function
    End Select

    ' Fallback to the form's Item Type when Cost Group is not populated or not present.
    ItemType = UCase$(Trim$(GetFirstNonBlankFieldTextLegacyTypes(rsSource, _
        Array("Item Type", "ItemType", "FG Item Type"))))

    If InStr(1, ItemType, "ASSEMBLY LEVEL", vbTextCompare) > 0 Or ItemType = "ASSY" Then
        GetFinishedGoodsCostGroupLegacyTypes = "AL"
        Exit Function
    End If

    If InStr(1, ItemType, "SANDWICH LEVEL", vbTextCompare) > 0 Or ItemType = "SW" Then
        GetFinishedGoodsCostGroupLegacyTypes = "SW"
        Exit Function
    End If

    ' Last fallback: AX Item Group uses labels such as "Assembly Level - 3200"
    ' and "Sandwich Level - 3100". Other item groups are treated as FG.
    AXItemGroup = UCase$(Trim$(GetFirstNonBlankFieldTextLegacyTypes(rsSource, _
        Array("AX Item Group", "AXItemGroup"))))

    If AXItemGroup = "AL" Or InStr(1, AXItemGroup, "ASSEMBLY LEVEL", vbTextCompare) > 0 _
        Or AXItemGroup = "3200" Or AXItemGroup = "03200" Then
        GetFinishedGoodsCostGroupLegacyTypes = "AL"
    ElseIf AXItemGroup = "SW" Or InStr(1, AXItemGroup, "SANDWICH LEVEL", vbTextCompare) > 0 _
        Or AXItemGroup = "3100" Or AXItemGroup = "03100" Then
        GetFinishedGoodsCostGroupLegacyTypes = "SW"
    Else
        ' The remaining Finished Goods requests use the standard FG convention.
        GetFinishedGoodsCostGroupLegacyTypes = "FG"
    End If

End Function


Private Function GetFirstNonBlankFieldTextLegacyTypes(ByVal rs As DAO.Recordset, ByVal fieldNames As Variant) As String

    Dim fieldName As Variant
    Dim fieldValue As String

    For Each fieldName In fieldNames
        fieldValue = Trim$(GetFieldTextLegacyTypes(rs, CStr(fieldName)))
        If Len(fieldValue) > 0 Then
            GetFirstNonBlankFieldTextLegacyTypes = fieldValue
            Exit Function
        End If
    Next fieldName

End Function


Private Function CleanFinishedGoodsCustomerCodeLegacyTypes(ByVal customerCode As String) As String

    customerCode = Trim$(customerCode)

    Select Case UCase$(customerCode)
        Case "", "N/A", "NA", "NOT APPLICABLE", "NOT AVAILABLE", "NONE", "NULL", "0", "0.0"
            CleanFinishedGoodsCustomerCodeLegacyTypes = vbNullString
        Case Else
            CleanFinishedGoodsCustomerCodeLegacyTypes = customerCode
    End Select

End Function


Private Function RemoveLeadingCustomerNameLegacyTypes(ByVal descriptionText As String, ByVal customerName As String) As String

    descriptionText = Trim$(descriptionText)
    customerName = Trim$(customerName)

    If Len(customerName) > 0 And Len(descriptionText) >= Len(customerName) Then
        If StrComp(Left$(descriptionText, Len(customerName)), customerName, vbTextCompare) = 0 Then
            If Len(descriptionText) = Len(customerName) Then
                descriptionText = vbNullString
            ElseIf Mid$(descriptionText, Len(customerName) + 1, 1) = " " Then
                descriptionText = Trim$(Mid$(descriptionText, Len(customerName) + 1))
            End If
        End If
    End If

    RemoveLeadingCustomerNameLegacyTypes = descriptionText

End Function


Private Function AppendFinishedGoodsPartLegacyTypes(ByVal currentText As String, ByVal newPart As String) As String

    currentText = Trim$(currentText)
    newPart = Trim$(newPart)

    If Len(newPart) = 0 Then
        AppendFinishedGoodsPartLegacyTypes = currentText
    ElseIf Len(currentText) = 0 Then
        AppendFinishedGoodsPartLegacyTypes = newPart
    Else
        AppendFinishedGoodsPartLegacyTypes = currentText & " " & newPart
    End If

End Function


Private Function BuildRawMaterialDescription(ByVal rsSource As DAO.Recordset, ByVal originalDescription As String) As String

    Dim result As String
    Dim storageValue As String
    Dim welfareValue As String
    Dim allergenValues As Collection
    Dim allergenCodes As String

    result = Trim$(originalDescription)

    ' Read the actual SharePoint column name: Storage Temp.
    storageValue = GetFirstChoiceValueLegacyTypes(rsSource, Array("Storage Temp"))

    Select Case LCase$(Trim$(storageValue))
        Case "fresh"
            storageValue = "FR"
        Case "frozen", "frz"
            storageValue = "FZ"
        Case "ambient"
            storageValue = "AMB"
        Case "iqf"
            storageValue = "IQF"
        Case "intermediate bulk container"
            storageValue = "IBC"
        Case "tempered"
            storageValue = "TMP"
        Case Else
            ' Not Applicable, Fresh/Frz, blank and unknown values are omitted.
            storageValue = vbNullString
    End Select

    If Len(storageValue) > 0 Then
        result = AppendDescriptionPartLegacyTypes(result, storageValue)
    End If

    ' Read the actual SharePoint column name: Welfare Standard.
    ' Convert its display description to the abbreviation in abbs.xlsx.
    welfareValue = GetFirstChoiceValueLegacyTypes(rsSource, Array("Welfare Standard"))
    welfareValue = WelfareAbbreviationLegacyTypes(welfareValue)

    If Len(welfareValue) > 0 Then
        result = AppendDescriptionPartLegacyTypes(result, welfareValue)
    End If

    ' Species is intentionally ignored.

    ' Read the actual SharePoint column name: Allergens.
    ' Convert values to codes, deduplicate and sort alphabetically.
    Set allergenValues = GetChoiceValuesLegacyTypes(rsSource, Array("Allergens"))
    allergenCodes = AllergenCodesLegacyTypes(allergenValues)

    If Len(allergenCodes) > 0 Then
        result = AppendDescriptionPartLegacyTypes(result, "*" & allergenCodes & "*")
    End If

    BuildRawMaterialDescription = result

End Function

Private Function AppendDescriptionPartLegacyTypes(ByVal currentDescription As String, ByVal newPart As String) As String

    currentDescription = Trim$(currentDescription)
    newPart = Trim$(newPart)

    If Len(newPart) = 0 Then
        AppendDescriptionPartLegacyTypes = currentDescription
    ElseIf Len(currentDescription) = 0 Then
        AppendDescriptionPartLegacyTypes = newPart
    ElseIf InStr(1, " " & currentDescription & " ", " " & newPart & " ", vbTextCompare) > 0 Then
        ' Prevent adding the exact same suffix token more than once.
        AppendDescriptionPartLegacyTypes = currentDescription
    Else
        AppendDescriptionPartLegacyTypes = currentDescription & " " & newPart
    End If

End Function


Private Function GetFirstChoiceValueLegacyTypes(ByVal rs As DAO.Recordset, ByVal fieldNames As Variant) As String

    Dim values As Collection

    Set values = GetChoiceValuesLegacyTypes(rs, fieldNames)

    If values.Count > 0 Then
        GetFirstChoiceValueLegacyTypes = CStr(values(1))
    Else
        GetFirstChoiceValueLegacyTypes = vbNullString
    End If

End Function


Private Function GetChoiceValuesLegacyTypes(ByVal rs As DAO.Recordset, ByVal fieldNames As Variant) As Collection

    Dim results As Collection
    Dim fieldName As Variant
    Dim fieldValue As Variant
    Dim childRS As DAO.Recordset2
    Dim childValue As Variant

    Set results = New Collection

    For Each fieldName In fieldNames

        Set childRS = Nothing
        Err.Clear

        ' SharePoint multi-value fields are exposed by DAO as a child Recordset2.
        On Error Resume Next
        Set childRS = rs.Fields(CStr(fieldName)).value

        If Err.Number = 0 And Not childRS Is Nothing Then
            Err.Clear
            On Error GoTo 0

            Do While Not childRS.EOF
                childValue = childRS.Fields(0).value

                If Not IsNull(childValue) And Not IsEmpty(childValue) Then
                    If Len(Trim$(CStr(childValue))) > 0 Then
                        results.Add Trim$(CStr(childValue))
                    End If
                End If

                childRS.MoveNext
            Loop

            childRS.Close
            Set childRS = Nothing

        Else
            ' Single-value choice fields are returned as ordinary scalar values.
            Err.Clear
            Set childRS = Nothing
            On Error GoTo 0

            On Error Resume Next
            fieldValue = rs.Fields(CStr(fieldName)).value

            If Err.Number = 0 Then
                If Not IsNull(fieldValue) And Not IsEmpty(fieldValue) Then
                    If Len(Trim$(CStr(fieldValue))) > 0 Then
                        results.Add Trim$(CStr(fieldValue))
                    End If
                End If
            End If

            Err.Clear
            On Error GoTo 0
        End If

    Next fieldName

    Set GetChoiceValuesLegacyTypes = results

End Function


Private Function AllergenCodesLegacyTypes(ByVal allergenValues As Collection) As String

    Dim codes As Object
    Dim allergenValue As Variant
    Dim allergenText As String
    Dim allergenCode As String
    Dim codeArray() As String
    Dim key As Variant
    Dim i As Long
    Dim j As Long
    Dim temp As String
    Dim result As String

    Set codes = CreateObject("Scripting.Dictionary")
    codes.CompareMode = vbTextCompare

    For Each allergenValue In allergenValues

        allergenText = Trim$(CStr(allergenValue))
        allergenCode = AllergenCodeFromSharePointLegacyTypes(allergenText)

        If Len(allergenCode) > 0 Then
            If Not codes.exists(allergenCode) Then
                codes.Add allergenCode, True
            End If
        End If

    Next allergenValue

    If codes.Count = 0 Then
        AllergenCodesLegacyTypes = vbNullString
        Exit Function
    End If

    ReDim codeArray(0 To codes.Count - 1)

    i = 0
    For Each key In codes.Keys
        codeArray(i) = CStr(key)
        i = i + 1
    Next key

    ' Alphabetical sort, independent of the order SharePoint returns choices.
    For i = LBound(codeArray) To UBound(codeArray) - 1
        For j = i + 1 To UBound(codeArray)
            If StrComp(codeArray(i), codeArray(j), vbTextCompare) > 0 Then
                temp = codeArray(i)
                codeArray(i) = codeArray(j)
                codeArray(j) = temp
            End If
        Next j
    Next i

    For i = LBound(codeArray) To UBound(codeArray)
        If Len(result) > 0 Then result = result & "."
        result = result & codeArray(i)
    Next i

    AllergenCodesLegacyTypes = result

End Function


Private Function AllergenCodeFromSharePointLegacyTypes(ByVal allergenText As String) As String

    Dim cleanText As String
    Dim openPos As Long
    Dim closePos As Long
    Dim code As String

    cleanText = Trim$(allergenText)

    If Len(cleanText) = 0 Then Exit Function

    Select Case LCase$(cleanText)
        Case "not applicable", "n/a", "cereals"
            Exit Function

        Case "celery", "celery *ce*"
            code = "CE"
        Case "cereals containing gluten", "cereals containing gluten *gl*"
            code = "GL"
        Case "cereals containing wheat", "cereals containing wheat *wh*"
            code = "WH"
        Case "cereals containing rye"
            code = "RY"
        Case "cereals containing barley", "cereals containing barley *ba*"
            code = "BA"
        Case "cereals containing oats", "cereals containing oats *oa*"
            code = "OA"
        Case "crustaceans", "crustaceans *cr*", "crustacean"
            code = "CR"
        Case "eggs", "eggs *eg*", "egg"
            code = "EG"
        Case "fish", "fish *fi*"
            code = "FI"
        Case "lupin", "lupin *lu*"
            code = "LU"
        Case "milk", "milk *mi*"
            code = "MI"
        Case "molluscs", "molluscs *mo*"
            code = "MO"
        Case "mustard", "mustard *mu*"
            code = "MU"
        Case "nuts", "nuts *nu*"
            code = "NU"
        Case "peanuts", "peanuts *pe*"
            code = "PE"
        Case "sesame seeds", "sesame seeds *se*", "sesame"
            code = "SE"
        Case "soya", "soya *so*"
            code = "SO"
        Case "sulphur dioxide", "sulphur dioxide *su*", "sulphites"
            code = "SU"
        Case Else
            ' Also accept values already supplied in *XX* form.
            openPos = InStr(1, cleanText, "*", vbBinaryCompare)
            If openPos > 0 Then
                closePos = InStr(openPos + 1, cleanText, "*", vbBinaryCompare)
                If closePos > openPos Then
                    code = UCase$(Trim$(Mid$(cleanText, openPos + 1, closePos - openPos - 1)))
                End If
            End If
    End Select

    AllergenCodeFromSharePointLegacyTypes = UCase$(Trim$(code))

End Function
